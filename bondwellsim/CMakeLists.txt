cmake_minimum_required(VERSION 3.16)
project(bondwellsim C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CORE ../z80core)
add_library(bondwell STATIC
    srcsim/memory.c srcsim/io.c srcsim/fdc.c srcsim/video.c
    srcsim/keyboard.c srcsim/platform.c
    ${CORE}/simglb.c ${CORE}/simcore.c ${CORE}/simz80.c
    ${CORE}/simz80-cb.c ${CORE}/simz80-dd.c ${CORE}/simz80-ddcb.c
    ${CORE}/simz80-ed.c ${CORE}/simz80-fd.c ${CORE}/simz80-fdcb.c)
target_include_directories(bondwell PUBLIC srcsim ${CORE})
if(NOT WIN32)
    target_compile_definitions(bondwell PRIVATE _POSIX_C_SOURCE=200809L)
endif()
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(bondwell PRIVATE -Wall -Wextra)
endif()
add_executable(bondwellsim srcsim/main.c srcsim/ui.c)
target_link_libraries(bondwellsim PRIVATE bondwell)
if(WIN32)
    target_link_libraries(bondwellsim PRIVATE user32 gdi32)
else()
    find_package(SDL2 QUIET)
    if(SDL2_FOUND)
        target_compile_definitions(bondwellsim PRIVATE BW_SDL)
        target_include_directories(bondwellsim PRIVATE ${SDL2_INCLUDE_DIRS})
        target_link_libraries(bondwellsim PRIVATE ${SDL2_LIBRARIES})
    endif()
endif()
enable_testing()
add_executable(bondwell_tests tests/test_machine.c)
target_link_libraries(bondwell_tests PRIVATE bondwell)
add_test(NAME machine COMMAND bondwell_tests)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_FOUND)
    add_test(NAME disk_converters COMMAND ${Python3_EXECUTABLE} -m unittest discover
        -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p "test_*.py")
endif()
